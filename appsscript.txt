const FOLDER_ID = '1Sks47o6MQGUWzR-rIx3jXhkWjiw4wrxe';

function doPost(e) {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: "Сервер занят, повторите попытку" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // ========================================================================
    // ШАГ 1: ТЕКСТОВЫЕ ДАННЫЕ + СОХРАНЕНИЕ ID
    // ========================================================================
    if (data.action === "TEXT") {
      var rawName = (data.user && data.user.trim()) ? data.user.trim() : "Без имени";
      var sheetName = rawName.replace(/[\*\?\/\\\[\]:]/g, '_').substring(0, 50);
      var sheet = ss.getSheetByName(sheetName);

      if (!sheet) {
        sheet = ss.insertSheet(sheetName);
        sheet.appendRow([
          "Дата и время создания", "Торговая точка", "Показания счетчика",
          "Мелкие игрушки", "Заманухи", "Проброс", "Проблемы на маршруте", "Фото", "ID отчета"
        ]);
        sheet.getRange("A1:I1").setFontWeight("bold").setBackground("#e8f0fe");
        sheet.setFrozenRows(1);
        // Скрываем служебную колонку с ID, чтобы не мешала пользователям
        sheet.hideColumns(9);
      }

      sheet.appendRow([
        data.timestamp || new Date().toLocaleString("ru-RU"),
        data.pointName || "",
        data.meter || "",
        data.smallToys || 0,
        data.zamanukha || 0,
        data.probros || "",
        data.problems || "Нет",
        data.hasPhoto ? "Загрузка фото..." : "Без фото",
        data.id || "" // Колонка I: уникальный ID отчета с клиента
      ]);

      SpreadsheetApp.flush();
      return ContentService
        .createTextOutput(JSON.stringify({ status: "success", sheetName: sheetName }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // ========================================================================
    // ШАГ 2: СОХРАНЕНИЕ ФОТО С ПОИСКОМ СТРОКИ ПО ID
    // ========================================================================
    if (data.action === "PHOTO") {
      var targetSheet = ss.getSheetByName(data.sheetName);
      if (!targetSheet) {
        return ContentService
          .createTextOutput(JSON.stringify({ status: "error", message: "Лист не найден: " + data.sheetName }))
          .setMimeType(ContentService.MimeType.JSON);
      }

      // 1. ПОИСК ТОЧНОЙ СТРОКИ ПО УНИКАЛЬНОМУ ID В КОЛОНКЕ I
      var reportId = (data.id || "").trim();
      var targetRow = null;

      if (reportId) {
        var finder = targetSheet.getRange("I:I").createTextFinder(reportId).matchEntireCell(true);
        var foundCell = finder.findNext();
        if (foundCell) {
          targetRow = foundCell.getRow();
        }
      }

      // Если строка с таким ID не найдена — НЕЛЬЗЯ возвращать success!
      if (!targetRow) {
        return ContentService
          .createTextOutput(JSON.stringify({ 
            status: "error", 
            message: "Строка отчета не найдена в таблице (ID: " + reportId + "). Возможно, строка была удалена." 
          }))
          .setMimeType(ContentService.MimeType.JSON);
      }

      // 2. ЗАГРУЗКА ФАЙЛА НА GOOGLE ДИСК
      var photoUrl = "";
      var fileId = "";

      try {
        var dateStr = Utilities.formatDate(new Date(), "GMT+3", "yyyy-MM-dd_HH-mm-ss");
        var fileName = (data.sheetName || "Фото") + "_" + dateStr + ".jpg";

        var decoded = Utilities.base64Decode(data.photoBase64);
        var imageBlob = Utilities.newBlob(decoded, "image/jpeg", fileName);

        var uploadedFile = Drive.Files.create({
          name: fileName,
          mimeType: "image/jpeg",
          parents: [FOLDER_ID]
        }, imageBlob, { fields: "id, webViewLink" });

        fileId = uploadedFile.id;

        try {
          Drive.Permissions.create({
            role: "reader",
            type: "anyone"
          }, fileId);
        } catch(permErr) {}

        photoUrl = uploadedFile.webViewLink || ("https://drive.google.com/file/d/" + fileId + "/view");

      } catch (driveErr) {
        return ContentService
          .createTextOutput(JSON.stringify({ status: "error", message: "Ошибка Drive v3: " + driveErr.message }))
          .setMimeType(ContentService.MimeType.JSON);
      }

      // 3. ЗАПИСЬ ССЫЛКИ В ТОЧНО НАЙДЕННУЮ СТРОКУ
      try {
        targetSheet.getRange(targetRow, 8).setFormula('=HYPERLINK("' + photoUrl + '", "📷 Открыть фото")');
      } catch (formulaErr) {
        var richText = SpreadsheetApp.newRichTextValue()
          .setText("📷 Открыть фото")
          .setLinkUrl(photoUrl)
          .build();
        targetSheet.getRange(targetRow, 8).setRichTextValue(richText);
      }

      SpreadsheetApp.flush();

      return ContentService
        .createTextOutput(JSON.stringify({ status: "success", photoUrl: photoUrl }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: "Неизвестное действие" }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    try { lock.releaseLock(); } catch(e) {}
  }
}
